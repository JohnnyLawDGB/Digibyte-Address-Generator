# HoldFast Vault boot configuration for Raspberry Pi 3 B, 3 B+, 4 B and Pi 400.
# The boot ROM loads start.elf on a Pi 3 and start4.elf on a Pi 4 by itself.
kernel=Image
arm_64bit=1
disable_overscan=1

# Radios off at the firmware level as well as absent from the kernel.
dtoverlay=disable-wifi
dtoverlay=disable-bt

# No serial console: the only way in is the HDMI screen and a USB keyboard.
enable_uart=0
